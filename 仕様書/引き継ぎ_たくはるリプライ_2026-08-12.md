  ```bash

  cp build/app/outputs/flutter-apk/app-release.apk build/takuharu-finance-v0.2.107.apk
  gh release create takuharu-v0.2.107 apps/takuharu_finance/build/takuharu-finance-v0.2.107.apk \

  ```
